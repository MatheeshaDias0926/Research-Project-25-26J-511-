        Find the process using Port 3000
        In PowerShell, use this command to find the Process ID (PID):

        PowerShell
        netstat -ano | findstr :3000

        Example output:   TCP    0.0.0.0:3000           0.0.0.0:0              LISTENING       12345

        (In this case, 12345 is the PID).

        Kill the process
        Once you have that number (the PID), run this to force it to stop:

        PowerShell
        Stop-Process -Id [YOUR_PID] -Force

from http.server import HTTPServer, BaseHTTPRequestHandler
import json
import threading
import time
from urllib.parse import urlparse, parse_qs

# Data store
data_store = {
    "lat": 6.9271, 
    "lng": 79.8612,
    "mode": "real",
    "is_moving": False
}

# Fake movement logic
def fake_movement():
    while True:
        if data_store["mode"] == "fake":
            data_store["lat"] += 0.0001
            data_store["lng"] += 0.0001
        time.sleep(2)

class GPSServer(BaseHTTPRequestHandler):
    def do_GET(self):
        query = parse_qs(urlparse(self.path).query)
        if "setmode" in query:
            data_store["mode"] = query["setmode"][0]

        self.send_response(200)
        self.send_header('Content-type', 'application/json')
        self.send_header('Access-Control-Allow-Origin', '*')
        self.end_headers()
        
        response = json.dumps({
            "status": "success",
            "mode": data_store["mode"],
            "location": {"lat": data_store["lat"], "lng": data_store["lng"]}
        })
        self.wfile.write(response.encode())

if __name__ == "__main__":
    threading.Thread(target=fake_movement, daemon=True).start()
    HTTPServer(('', 8080), GPSServer).serve_forever()
